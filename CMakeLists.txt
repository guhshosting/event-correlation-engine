cmake_minimum_required(VERSION 3.20)
project(event-correlation-engine VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "This portable release requires an x64 build")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
enable_language(RC)
add_library(engine src/engine.cpp)
target_include_directories(engine PUBLIC src)
target_compile_definitions(engine PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(engine PUBLIC wevtapi xmllite shlwapi ole32)
add_executable(event-correlation-engine src/main.cpp src/version.rc)
target_link_libraries(event-correlation-engine PRIVATE engine)
if(MSVC)
  target_link_options(event-correlation-engine PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
  target_compile_options(engine PRIVATE /W4 /WX /analyze /permissive- /utf-8)
  target_compile_options(event-correlation-engine PRIVATE /W4 /WX /analyze /permissive- /utf-8)
endif()
enable_testing()
add_executable(engine-tests tests/tests.cpp)
target_link_libraries(engine-tests PRIVATE engine)
if(MSVC)
 target_compile_options(engine-tests PRIVATE /W4 /WX /analyze /permissive- /utf-8)
endif()
add_test(NAME engine COMMAND engine-tests)
add_test(NAME help COMMAND event-correlation-engine --help)
add_test(NAME version COMMAND event-correlation-engine --version)
add_test(NAME cli-json-contract COMMAND "${CMAKE_COMMAND}" "-DPROGRAM=$<TARGET_FILE:event-correlation-engine>" "-DFIXTURES=$<TARGET_FILE:engine-tests>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake")
option(ECE_LIVE_TESTS "Run read-only local logs with a restricted test token" OFF)
if(ECE_LIVE_TESTS)
 add_executable(live-tests tests/live.cpp)
 target_link_libraries(live-tests PRIVATE engine advapi32)
 if(MSVC)
  target_compile_options(live-tests PRIVATE /W4 /WX /analyze /permissive- /utf-8)
 endif()
 add_test(NAME restricted-live COMMAND live-tests)
endif()
install(TARGETS event-correlation-engine RUNTIME DESTINATION .)
install(FILES README.md LICENSE VALIDATION.md DESTINATION .)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "event-correlation-engine-1.0.0-windows-x64")
include(CPack)
